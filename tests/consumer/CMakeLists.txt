cmake_minimum_required(VERSION 3.16)
project(talon_status_consumer LANGUAGES CXX)
if(TALON_STATUS_CONSUMER_MODE STREQUAL "subdirectory")
  add_subdirectory("${TALON_STATUS_SOURCE_DIR}" talon-status EXCLUDE_FROM_ALL)
elseif(TALON_STATUS_CONSUMER_ABSL)
  find_package(talon-status 0.1 CONFIG REQUIRED COMPONENTS absl)
else()
  find_package(talon-status 0.1 CONFIG REQUIRED)
endif()
add_executable(consumer main.cc helper.cc)
target_link_libraries(consumer PRIVATE talon::status)
enable_testing()
add_test(NAME consumer.run COMMAND consumer)
if(TALON_STATUS_CONSUMER_ABSL)
  add_executable(consumer_absl main_absl.cc)
  target_link_libraries(consumer_absl PRIVATE talon::status_absl)
  add_test(NAME consumer.absl COMMAND consumer_absl)
endif()
