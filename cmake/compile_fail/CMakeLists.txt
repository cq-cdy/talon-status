cmake_minimum_required(VERSION 3.16)
project(talon_compile_fail LANGUAGES CXX)
add_executable(baseline baseline.cc)
add_executable(rejected EXCLUDE_FROM_ALL "${TALON_TEST_SOURCE}")
target_include_directories(rejected PRIVATE "${TALON_INCLUDE_DIR}")
target_compile_definitions(rejected PRIVATE TALON_STATUS_BACKEND=${TALON_BACKEND_VALUE})
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
  target_compile_options(baseline PRIVATE /Zc:preprocessor)
  if(TALON_TRADITIONAL_MSVC)
    target_compile_options(rejected PRIVATE /Zc:preprocessor-)
  else()
    target_compile_options(rejected PRIVATE /Zc:preprocessor)
  endif()
endif()
if(TALON_NO_EXCEPTIONS)
  if(MSVC)
    target_compile_options(baseline PRIVATE /EHs-c-)
    target_compile_options(rejected PRIVATE /EHs-c-)
    target_compile_definitions(baseline PRIVATE _HAS_EXCEPTIONS=0)
    target_compile_definitions(rejected PRIVATE _HAS_EXCEPTIONS=0)
  else()
    target_compile_options(baseline PRIVATE -fno-exceptions)
    target_compile_options(rejected PRIVATE -fno-exceptions)
  endif()
endif()
